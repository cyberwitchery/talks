  \section{recap}
  % manual -> source of truth + small automations -> glue in the middle
  % ==================================================================
  \begin{frame}{by hand}
    \begin{itemize}
      \item ssh into the box, type the config.
      \item fat-finger something, find out later.
      \item forget what we did and why.
      \item learn how to rescue things really well.
    \end{itemize}
  \end{frame}

  \begin{frame}{write it down and script it}
    \begin{itemize}
			\item document the setup in a \tealaccent{source of truth} (often >1).
      \item write small automations against it using salt, ansible, templates.
      \item \textcolor{cwFaint}{this is where most of us still are today.}
    \end{itemize}
  \end{frame}

  \begin{frame}{what that left us}
    \vfill
    \begin{center}
      \begin{tikzpicture}[node distance=0.4cm]
        \node[sys] (a) {netbox};

        \node[sys, right=of b] (c) {containerlab};
        \node[sys, right=of c] (d) {prometheus};
        \node[sys, right=of d] (e) {dns};
      \end{tikzpicture}
    \end{center}
    \vskip6pt
    \begin{itemize}
      \item bespoke code in the middle.
      \item too many systems that don't talk to each other.
      \item our model has to fit the vendor world.
    \end{itemize}
    \vfill
  \end{frame}

  \section{a model of our own}
  % why we end up with one anyway
  % ==================================================================
  \begin{frame}{the model we actually work from}
    \begin{itemize}
      \item the source of truth's object model,
      \item plus custom fields and plugins,
      \item plus the spreadsheet on a samba share,
      \item plus concepts that only live in people's heads.
    \end{itemize}
    \subhero{i think most of us end up with our own model anyway.}
  \end{frame}

  \begin{frame}[fragile]{an example}
    \vfill
\begin{lstlisting}
dcim.device:
  key: { name: { type: slug } }
  fields:
    name:  { type: slug }
    kind:  { type: string }
    image: { type: string }
    cmd:   { type: string }
\end{lstlisting}
    \vskip6pt
    \subhero{a lab node needs a kind, an image, and a command. netbox has none
    of them, so they end up in custom fields, config contexts, or a script.}
    \vfill
  \end{frame}

  \begin{frame}{the model is already there}
    \hero{every integration already decides what devices to look at,
    which fields matter, and \tealaccent{what gets dropped}.}
    \subhero{the model exists somewhere in the aether.}
  \end{frame}

  \section{the model in the middle}
  % n x m -> n, every operation is a transform
  % ==================================================================
  \begin{frame}{a different center}
    \hero{put \tealaccent{your data model} in the middle,
    and transform every system into and out of it.}
  \end{frame}

  \begin{frame}{one transform per system}
    \vfill
    \begin{center}
      \begin{tikzpicture}[dot/.style={circle, fill=cwBody, inner sep=1.5pt}]
        \begin{scope}
          \foreach \i in {0,...,5} \coordinate (l\i) at (90+60*\i:1.3);
          \foreach \i in {0,...,4} {
            \pgfmathtruncatemacro{\n}{\i+1}
            \foreach \j in {\n,...,5} \draw[edge] (l\i) -- (l\j);
          }
          \foreach \i in {0,...,5} \node[dot] at (l\i) {};
          \node[font=\scriptsize, text=cwMuted] at (0,-1.9) {point to point: n $\times$ m};
        \end{scope}
        \begin{scope}[xshift=6cm]
          \foreach \i in {0,...,5} \coordinate (r\i) at (90+60*\i:1.3);
          \foreach \i in {0,...,5} \draw[edge] (0,0) -- (r\i);
          \foreach \i in {0,...,5} \node[dot] at (r\i) {};
          \node[fill=cwGround, text=cwTeal, font=\scriptsize\bfseries,
            inner sep=3pt] at (0,0) {model};
          \node[font=\scriptsize, text=cwMuted] at (0,-1.9) {model in the middle: n};
        \end{scope}
      \end{tikzpicture}
    \end{center}
		\subhero{a new system is one transform (or two),
    not one per system it talks to.}
    \vfill
  \end{frame}

  \begin{frame}{every operation is a transform}
    \begin{itemize}
      \item \textbf{deploy}: netbox $\to$ containerlab, ansible, prometheus.
			\item \textbf{drift}: containerlab $\to$ netbox (just check).
			\item \textbf{migration}: netbox $\to$ nautobot. (you can even run both for a while,
				then cut over.)
    \end{itemize}
    \subhero{it’s all the same thing!}
  \end{frame}

  \begin{frame}{a compiler for your infra}
    \hero{many frontends, many backends.\\[0.3em]
    the part that unifies them is \tealaccent{the data and its types}.}
    \subhero{what you need is a system that is good at defining data models
    and transforming between them.}
  \end{frame}

  \section{trade-offs}
  % you own the engineering
  % ==================================================================
  \begin{frame}{you own the engineering}
    \begin{itemize}
      \item[\textcolor{cwFaint}{\textbar}] \textcolor{cwMuted}{all in on a vendor:
        excellent interoperability out of the box as long as you color inside
        the lines.}
      \item[\textcolor{cwTeal}{\textbar}] own the model, make it the shape you
        want, and \tealaccent{more decisions that you own}.
      \item[\textcolor{cwFaint}{\textbar}] with great power comes great
        responsibility.
    \end{itemize}
  \end{frame}

  \begin{frame}{reinventing the universe}
    \hero{a bespoke model can mean\\[0.3em]
    \tealaccent{modelling everything from scratch}.}
    \subhero{expressiveness and recipes help, but teams can get stuck in
    design limbo.}
  \end{frame}

  \begin{frame}{it only pays off at a certain size}
    \hero{that size gets \tealaccent{smaller} as tools get better.}
    \subhero{it’s never zero.}
  \end{frame}

  \begin{frame}{takeaways}
    \begin{itemize}
			\item if you own the model, the transforms get easier.
      \item don’t focus on the boxes, focus on \tealaccent{the edges between them}.
			\item think about the shape of your data, and don’t take anything as
				gospel.
		\end{itemize}
  \end{frame}
